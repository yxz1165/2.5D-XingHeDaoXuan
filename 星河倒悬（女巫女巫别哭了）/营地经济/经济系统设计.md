# 经济系统——卦术世界的货币与交易

> 不做现实经济模拟。这个世界的钱只有两种：你从敌人身上搜出来的铜钱，和你从星君遗迹里挖出来的星晶。简单直接。

---

## 一、两种钱

| | 铜钱 | 星晶 |
|------|------|------|
| 本质 | 人间流通·谁都有 | 星君遗留·太史阁在搜刮 |
| 来源 | 打敌人掉·卖破烂·任务 | BOSS·隐藏宝箱·互卦深层 |
| 用途 | 买材料·锻造·符纸·传送 | 高端锻造·紫符·稀有丹药 |
| 兑换 | — | 不可换·不可买 |

就两种。不需要第三种。

---

## 二、钱从哪来

打敌人掉铜钱——这是最主要来源。不需要"采集-加工-出售"的循环。杀敌拿钱，直接。

| 敌人 | 掉多少 |
|------|:--:|
| 杂兵 | 5-20文 |
| 精英 | 30-80文 |
| BOSS | 100-300文 + 2-5星晶（寿星宫商徵+国师最多7星晶） |

特殊收入：太史阁敌人掉的钱比海溟暗卫多（太史阁有钱）。暗卫掉的材料比太史阁值钱（海溟材料稀有）。玩家自己选——刷钱去打太史阁，刷材料去打海溟。

---

## 五、前期穷·后期富

玩家不需要"理财"。正常玩，经济曲线自然形成：

```
序章→娵訾：穷。买完青铜锭剩下几文钱。锻造+3是极限。
降娄→实沉：开始有余。铜钱剑要花108文铸造（很便宜）。
鹑首→大火：富了。但锻造+7要星晶——有钱也买不到。
析木→寿星：最终。锻造+9花光所有积蓄。但值。
```

**那个瞬间：** 实沉宫通关后，打完尉迟疆，捡起他掉的300文。打开背包发现铜钱第一次超过10000——"我有钱了。"

### 8.1 货币类型

```csharp
/// <summary>
/// 货币类型——与共享接口定义完全一致。仅两种，无第三种。
/// </summary>
public enum CurrencyType {
    CopperCoin = 0,  // 铜钱——人间流通·主要货币
    StarCrystal = 1  // 星晶——星君遗留·稀有·不可兑换
}

/// <summary>
/// 玩家钱包——运行时数据
/// </summary>
[Serializable]
public class Wallet {
    public int copperCoin;     // 铜钱数量
    public int starCrystal;    // 星晶数量
    
    // 基础操作
    public bool CanAfford(CurrencyType type, int amount) {
        return type switch {
            CurrencyType.CopperCoin => copperCoin >= amount,
            CurrencyType.StarCrystal => starCrystal >= amount,
            _ => false
        };
    }
    
    public bool Spend(CurrencyType type, int amount) {
        if (!CanAfford(type, amount)) return false;
        switch (type) {
            case CurrencyType.CopperCoin: copperCoin -= amount; break;
            case CurrencyType.StarCrystal: starCrystal -= amount; break;
        }
        return true;
    }
    
    public void Earn(CurrencyType type, int amount) {
        switch (type) {
            case CurrencyType.CopperCoin: copperCoin += amount; break;
            case CurrencyType.StarCrystal: starCrystal += amount; break;
        }
    }
}
```

### 8.2 掉落表（DropTable）

```csharp
/// <summary>
/// 敌人掉落条目
/// </summary>
[Serializable]
public class DropEntry {
    public CurrencyType currencyType;  // 铜钱 或 星晶
    public int minAmount;              // 最少掉落
    public int maxAmount;              // 最多掉落
    public float dropChance;           // 掉落概率（0-1）
    public int itemId;                 // 物品ID（0=仅货币）
    public int itemMinCount;           // 物品最少数量
    public int itemMaxCount;           // 物品最多数量
}

/// <summary>
/// 敌人类型掉落配置（ScriptableObject）
/// </summary>
[CreateAssetMenu(menuName = "星河倒悬/掉落表")]
public class DropTable : ScriptableObject {
    public string enemyTypeName;          // "太史学徒" / "冰原狼" / "焦烈(BOSS)"
    public List<DropEntry> guaranteedDrops;  // 必掉（dropChance=1.0）
    public List<DropEntry> randomDrops;      // 概率掉落
    
    // 计算实际掉落
    public (int copper, int crystal, List<int> itemIds, List<int> itemCounts) CalculateDrop() {
        int copper = 0, crystal = 0;
        var itemIds = new List<int>();
        var itemCounts = new List<int>();
        
        // 必掉
        foreach (var entry in guaranteedDrops) {
            int amount = Random.Range(entry.minAmount, entry.maxAmount + 1);
            AddDrop(entry, amount, ref copper, ref crystal, itemIds, itemCounts);
        }
        // 概率掉落
        foreach (var entry in randomDrops) {
            if (Random.value < entry.dropChance) {
                int amount = Random.Range(entry.minAmount, entry.maxAmount + 1);
                AddDrop(entry, amount, ref copper, ref crystal, itemIds, itemCounts);
            }
        }
        return (copper, crystal, itemIds, itemCounts);
    }
    
    private void AddDrop(DropEntry entry, int amount, ref int copper, ref int crystal,
                          List<int> itemIds, List<int> itemCounts) {
        if (entry.currencyType == CurrencyType.CopperCoin) copper += amount;
        else crystal += amount;
        if (entry.itemId > 0) {
            itemIds.Add(entry.itemId);
            itemCounts.Add(Random.Range(entry.itemMinCount, entry.itemMaxCount + 1));
        }
    }
}

/// <summary>
/// 按敌人难度预设的标准掉落（可通过DropTable覆盖）
/// </summary>
public static class StandardDropPresets {
    public static (int copperMin, int copperMax, int crystalMin, int crystalMax) GetPreset(EnemyTier tier) {
        return tier switch {
            EnemyTier.Mob     => (5, 20, 0, 0),      // 杂兵：5-20文
            EnemyTier.Elite   => (30, 80, 0, 0),     // 精英：30-80文
            EnemyTier.Boss    => (100, 300, 1, 3),   // BOSS：100-300文 + 1-3星晶
            EnemyTier.SuperBoss => (300, 800, 3, 8), // 超级BOSS
            _ => (5, 20, 0, 0)
        };
    }
}

public enum EnemyTier { Mob, Elite, Boss, SuperBoss }
```

### 8.3 商人/商店接口

```csharp
/// <summary>
/// 12道宫商人配置
/// </summary>
[CreateAssetMenu(menuName = "星河倒悬/商人配置")]
public class MerchantConfig : ScriptableObject {
    public string merchantName;          // "北部铁匠"
    public int palaceIndex;              // 0-11（对应12道宫）
    public float basePriceCoefficient;   // 基础物价系数（1.0-2.5）
    public List<ShopItem> inventory;     // 商品列表
}

/// <summary>
/// 商店商品条目
/// </summary>
[Serializable]
public class ShopItem {
    public int itemId;
    public string itemName;
    public CurrencyType currencyType;    // 售价货币类型
    public int basePrice;                // 基础价格（物价系数=1.0时的价格）
    public int stockLimit;               // 库存上限（-1=无限）
    public bool restockOnRest;           // 休息后补货
}

/// <summary>
/// 物价计算——综合命盘+道宫+好感度
/// </summary>
public static class PriceCalculator {
    // 12道宫基础物价系数
    public static readonly float[] PalacePriceCoefficients = {
        1.0f, 1.1f, 1.2f, 1.3f, 1.4f, 1.5f,  // 星纪→实沉
        1.6f, 1.7f, 1.8f, 2.0f, 2.2f, 2.5f   // 鹑首→寿星
    };
    
    /// <summary>
    /// 计算实际购买价格
    /// </summary>
    /// <param name="basePrice">商品基础价格</param>
    /// <param name="palaceIndex">当前所在道宫(0-11)</param>
    /// <param name="destiny">玩家命盘（财帛宫影响）</param>
    /// <param name="affinityLevel">与该商人的好感等级(0-5)</param>
    /// <returns>最终价格（铜钱）</returns>
    public static int CalculateBuyPrice(int basePrice, int palaceIndex, 
                                         ZiweiDestiny destiny, int affinityLevel) {
        float coefficient = PalacePriceCoefficients[Mathf.Clamp(palaceIndex, 0, 11)];
        
        // 财帛宫命盘修正
        float caiBoMod = GetCaiBoModifier(destiny);
        coefficient *= caiBoMod;
        
        // 好感折扣（每级-3%，最高-15%）
        float affinityDiscount = 1.0f - (affinityLevel * 0.03f);
        coefficient *= affinityDiscount;
        
        return Mathf.Max(1, Mathf.RoundToInt(basePrice * coefficient));
    }
    
    /// <summary>
    /// 财帛宫主星→购买折扣
    /// </summary>
    private static float GetCaiBoModifier(ZiweiDestiny destiny) {
        var star = destiny.palaceStars[(int)GongWei.CaiBo];
        // 检查四化：化禄=×0.5（半价），化忌=×1.5（涨价50%）
        float baseMod = star switch {
            ZiweiStar.WuQu => 0.80f,    // 武曲：买价-20%
            ZiweiStar.TaiYin => 1.00f,  // 太阴：买价不变·卖价+30%
            ZiweiStar.ZiWei => 1.00f,   // 紫微：买价不变·精英掉落×1.5
            ZiweiStar.TianFu => 0.90f,  // 天府：丹药材料-1
            ZiweiStar.QiSha => 1.00f,   // 七杀：BOSS掉落×1.3
            _ => 1.00f
        };
        
        // 四化修正
        var siHua = SiHuaRotator.GetSiHuaAtPalace(destiny, GongWei.CaiBo);
        if (siHua == SiHua.HuaLu) baseMod *= 0.5f;     // 化禄：半价
        if (siHua == SiHua.HuaJi) baseMod *= 1.5f;     // 化忌：涨价50%
        
        return baseMod;
    }
    
    /// <summary>
    /// 计算出售价格（玩家卖给商人）
    /// </summary>
    public static int CalculateSellPrice(int itemBaseValue, int palaceIndex, ZiweiDestiny destiny) {
        float coefficient = 0.3f; // 基础：三折回收
        coefficient *= PalacePriceCoefficients[Mathf.Clamp(palaceIndex, 0, 11)];
        
        // 太阴坐财帛→卖价+30%
        var star = destiny.palaceStars[(int)GongWei.CaiBo];
        if (star == ZiweiStar.TaiYin) coefficient *= 1.30f;
        
        return Mathf.Max(1, Mathf.RoundToInt(itemBaseValue * coefficient));
    }
}
```

### 8.4 经济事件管理器

```csharp
/// <summary>
/// 经济系统运行时管理器——统一管理所有经济操作
/// </summary>
public class EconomyManager : MonoBehaviour {
    public static EconomyManager Instance { get; private set; }
    
    [SerializeField] private Wallet playerWallet;
    
    // === 事件（供UI/音效系统订阅） ===
    public event System.Action<CurrencyType, int> OnCurrencyChanged;
    public event System.Action<int, int> OnPurchase;       // itemId, finalPrice
    public event System.Action<int, int, List<int>> OnEnemyLooted; // copper, crystal, itemIds
    
    // === 核心操作 ===
    
    /// <summary>击杀敌人→掉落金钱和物品</summary>
    public void ProcessEnemyLoot(DropTable dropTable, ZiweiDestiny destiny, int palaceIndex) {
        var (copper, crystal, itemIds, itemCounts) = dropTable.CalculateDrop();
        
        // 财帛宫掉落加成
        var star = destiny.palaceStars[(int)GongWei.CaiBo];
        if (star == ZiweiStar.ZiWei) copper = Mathf.RoundToInt(copper * 1.5f); // 紫微：精英掉落×1.5
        if (star == ZiweiStar.QiSha) crystal = Mathf.RoundToInt(crystal * 1.3f); // 七杀：BOSS掉落×1.3
        
        playerWallet.Earn(CurrencyType.CopperCoin, copper);
        playerWallet.Earn(CurrencyType.StarCrystal, crystal);
        
        OnCurrencyChanged?.Invoke(CurrencyType.CopperCoin, copper);
        OnCurrencyChanged?.Invoke(CurrencyType.StarCrystal, crystal);
        OnEnemyLooted?.Invoke(copper, crystal, itemIds);
    }
    
    /// <summary>从商人购买物品</summary>
    public bool PurchaseItem(ShopItem item, MerchantConfig merchant, 
                              ZiweiDestiny destiny, int affinityLevel) {
        int finalPrice = PriceCalculator.CalculateBuyPrice(
            item.basePrice, merchant.palaceIndex, destiny, affinityLevel);
        
        if (!playerWallet.CanAfford(item.currencyType, finalPrice))
            return false;
        
        playerWallet.Spend(item.currencyType, finalPrice);
        OnPurchase?.Invoke(item.itemId, finalPrice);
        return true;
    }
    
    /// <summary>星碑传送费用</summary>
    public bool TryFastTravel(int targetPalaceIndex) {
        const int TRAVEL_COST = 50; // 50文固定
        if (!playerWallet.CanAfford(CurrencyType.CopperCoin, TRAVEL_COST))
            return false;
        playerWallet.Spend(CurrencyType.CopperCoin, TRAVEL_COST);
        return true;
    }
    
    /// <summary>获取当前铜钱/星晶数量</summary>
    public int GetCopper() => playerWallet.copperCoin;
    public int GetStarCrystal() => playerWallet.starCrystal;
}
```

### 8.5 数据流概览

```
击杀敌人 → DropTable.CalculateDrop()
  → EconomyManager.ProcessEnemyLoot()
    → 财帛宫修正（紫微/七杀）
    → Wallet.Earn()
    → UI更新事件

商人购买 → PriceCalculator.CalculateBuyPrice()
  → 物价系数 × 财帛宫修正 × 好感折扣
  → EconomyManager.PurchaseItem()
    → Wallet.Spend()
    → 库存扣减事件

星碑传送 → EconomyManager.TryFastTravel()
  → 50文固定 → Wallet.Spend()
```
