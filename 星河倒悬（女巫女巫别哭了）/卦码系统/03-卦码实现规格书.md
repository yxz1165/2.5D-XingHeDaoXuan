// 公式：纳音ID = (天干ID + 地支ID) % 30
// 注意：这是简化映射——实际纳音有固定的天干地支配对表
// 游戏中纳音是铭刻后的"结果"而非"输入"——玩家铭刻了某个天干地支组合后自动获得对应纳音
public static int CalculateNaYinId(int tianGanId, int diZhiId) {
    // 天干地支各10/12 → 60种组合 → 30种纳音（每种纳音对应2个相邻干支组合）
    int comboIndex = (tianGanId * 12 + diZhiId) % 60;
    return comboIndex / 2; // 每2个连续组合=1种纳音
}

## 三、编码算法——完整C#实现

### 3.1 主入口

```csharp
using System;
using System.Collections.Generic;
using System.Text;

public class HexagramCodec {
    
    // === 常量 ===
    private const int TOTAL_BITS = 276; // 273有效bit + 3bit填充 = 276bit = 46组×6bit
    private const int CHARS_PER_GROUP = 6; // 每6bit=1个六十四卦字符
    private const string SEPARATOR = "☷";  // 坤卦=段分隔符
    private const int CURRENT_MAJOR_VERSION = 1;
    private const int CURRENT_MINOR_VERSION = 0;
    
    // === 编码入口 ===
    public static EncodeResult Encode(BuildData build, GameMode mode) {
        var bits = new BitWriter(TOTAL_BITS);
        
        // 段0：版本+模式 (6bit)
        bits.Write(CURRENT_MAJOR_VERSION, 2);
        bits.Write(CURRENT_MINOR_VERSION, 2);
        bits.Write((int)mode, 2);
        
        // 段1：法器段 (15bit)
        EncodeWeaponSegment(bits, build);
        
        // 段2：卦术段 (34bit max)
        EncodeTrigramSegment(bits, build);
        
        // 段3：铭刻段 (66bit)
        EncodeEngravingSegment(bits, build);
        
        // 段4：命盘段 (128bit)
        EncodeDestinySegment(bits, build);
        
        // 段5：同伴+丹鼎段 (20bit)
        EncodeCompanionSegment(bits, build);
        
        // 段6：校验+填充 (4bit校验·剩余自动填0)
        byte[] rawBits = bits.ToBytes();
        int checksum = CalculateChecksum(rawBits);
        bits.Write(checksum, 4);
        
        // 填充剩余bit到6的倍数
        int remainingBits = TOTAL_BITS - bits.Position;
        if (remainingBits > 0) bits.Write(0, remainingBits);
        
        // 渲染
        byte[] finalBits = bits.ToBytes();
        string hexagramCode = RenderHexagramCode(finalBits);
        string base64Code = RenderBase64Code(finalBits);
        
        return new EncodeResult {
            HexagramCode = hexagramCode,
            Base64Code = base64Code,
            Build = build
        };
    }
    
    // === 渲染为八卦码 ===
    private static string RenderHexagramCode(byte[] bits) {
        var sb = new StringBuilder();
        int groupCount = bits.Length * 8 / 6;
        
        // 段边界（1-based·在该组后插入☷分隔符）
        // 段0=1·段1=4·段2(卦术)=10·段3(铭刻)=21·段4(命盘)=42·段5(同伴)=45·段6(校验)=46
        int[] segmentEnds = { 1, 4, 10, 21, 42, 45, 46 };
        
        int segIndex = 0;
        for (int i = 0; i < groupCount; i++) {
            // 提取6bit
            int byteIndex = i * 6 / 8;
            int bitOffset = (i * 6) % 8;
            int value;
            
            if (bitOffset <= 2) {
                value = (bits[byteIndex] >> (2 - bitOffset)) & 0x3F;
            } else {
                value = ((bits[byteIndex] << (bitOffset - 2)) | 
                         (byteIndex + 1 < bits.Length ? bits[byteIndex + 1] >> (10 - bitOffset) : 0)) & 0x3F;
            }
            
            sb.Append(HEXAGRAM_UNICODE[value]);
            
            // 在段边界后插入分隔符
            if (segIndex < segmentEnds.Length - 1 && i + 1 == segmentEnds[segIndex]) {
                sb.Append(SEPARATOR);
                segIndex++;
            }
        }
        
        return sb.ToString();
    }
}
```

    private List<byte> _bytes = new List<byte>();
    private int _bitPos = 0; // 当前写入位置（全局bit偏移）
    
    public BitWriter(int totalBits) {
        int byteCount = (totalBits + 7) / 8;
        for (int i = 0; i < byteCount; i++) _bytes.Add(0);
    }
    
    public int Position => _bitPos;
    
    public void Write(int value, int bitCount) {
        if (bitCount <= 0 || bitCount > 32) throw new ArgumentException();
        
        // 清除高位
        value &= (1 << bitCount) - 1;
        
        for (int i = bitCount - 1; i >= 0; i--) {
            int byteIndex = _bitPos / 8;
            int bitInByte = 7 - (_bitPos % 8); // 高位在前
            
            if (byteIndex < _bytes.Count) {
                if ((value & (1 << i)) != 0) {
                    _bytes[byteIndex] |= (byte)(1 << bitInByte);
                }
            }
            _bitPos++;
        }
    }
    
    public byte[] ToBytes() => _bytes.ToArray();
}

public class BitReader {
    private byte[] _bytes;
    private int _bitPos = 0;
    
    public BitReader(byte[] bytes) { _bytes = bytes; }
    
    public int Read(int bitCount) {
        int value = 0;
        for (int i = 0; i < bitCount; i++) {
            int byteIndex = _bitPos / 8;
            int bitInByte = 7 - (_bitPos % 8);
            if (byteIndex < _bytes.Length) {
                if ((_bytes[byteIndex] & (1 << bitInByte)) != 0) {
                    value |= (1 << (bitCount - 1 - i));
                }
            }
            _bitPos++;
        }
        return value;
    }
}

## 六、校验算法

### 6.1 加权和校验（主力校验算法）

```csharp
public static int CalculateChecksum(byte[] bits) {
    // 取所有有效数据bit（不含校验位本身的最后4bit）
    // 按6bit一组分组·每组的值×组序号（1-based）·加权求和·取低4位
    int totalGroups = (bits.Length * 8 - 4) / 6; // 排除校验位
    int checksum = 0;
    
    for (int i = 0; i < totalGroups; i++) {
        int value = Extract6Bits(bits, i);
        checksum += value * (i + 1); // 位置加权
    }
    
    return checksum & 0xF; // 低4位
}

private static int Extract6Bits(byte[] bits, int groupIndex) {
    int bitStart = groupIndex * 6;
    int byteIndex = bitStart / 8;
    int bitOffset = bitStart % 8;
    
    if (bitOffset <= 2) {
        return (bits[byteIndex] >> (2 - bitOffset)) & 0x3F;
    } else {
        int val = (bits[byteIndex] << (bitOffset - 2)) & 0x3F;
        if (byteIndex + 1 < bits.Length) {
            val |= (bits[byteIndex + 1] >> (10 - bitOffset)) & 0x3F;
        }
        return val & 0x3F;
    }
}

public static bool VerifyChecksum(byte[] bits) {
    int totalGroups = (bits.Length * 8 - 4) / 6;
    int storedChecksum = 0;
    // 校验值存储在最后4bit
    int lastByteIndex = bits.Length - 1;
    storedChecksum = bits[lastByteIndex] & 0xF;
    
    int computed = CalculateChecksum(bits);
    return computed == storedChecksum;
}
```

# v2.0 轻量卦码（2026-08-10 新增·未取代v1.0）
# ══════════════════════════════════════════════

## 设计理念

**v1.0：** 52字符·46卦符+6☷分隔·273bit·信息完整但太长·像序列号

**v2.0：** 9卦一排·54bit·像占卜结果·改一个配件全码皆变

```
v1.0: ䷀䷝䷜䷲䷸䷖䷁䷊☷䷋䷌䷍䷎䷏䷐䷑☷䷒䷓䷔...
      → 52字符·无法一眼读完·结构化编码·改一处只变一个卦

v2.0: ䷝ ䷍ ䷁ ䷱ ䷾ ䷿ ䷌ ䷊ ䷀
      → 9卦一排·像占卜结果·哈希编码·改一处全码皆变
```

**两种格式共存。** v1.0用于内部存储·v2.0用于对外分享和展示。

---

## v2.0 编码规则

### 一、总览

```
输入：Build配置（约51bit有效数据）
  ├── 上卦4槽 C(8,4)×4! = 1680 → 11bit（排列编码·非独立编码）
  ├── 下卦4槽 C(8,4)×4! = 1680 → 11bit
  ├── 术数3槽 C(12,3) = 220 → 8bit
  ├── 法器2槽 C(4,2) = 6 → 3bit
  ├── 卦阵2槽 C(8,2) = 28 → 5bit
  ├── 同伴2槽 C(12,2) = 66 → 7bit
  └── 版本号 → 2bit
      ────────────
      47bit（仅核心Build·外观不进卦码）

流程：
  Build数据(47bit) → 加盐哈希 → 输出54bit → 切成9段(每段6bit)
  → 每段映射到1个六十四卦符 → 输出9个卦符
```

### 二、为什么用哈希

```
结构化编码（v1.0）：
  ䷝(上卦) ䷍(法器) ䷁(卦阵) ...
  改了法器 → 只有法器那个卦变了 → 其余8个一模一样
  
  问题：两个相似Build的码几乎一样 → 没有"码"的感觉

哈希编码（v2.0）：
  改任何一个配置 → 哈希结果完全不同 → 9个卦全部改变
  
  效果：和三角洲改枪码一样 → 改一个配件·整串码面目全非
  
  即使两个Build只差"把离卦换成震卦"：
    Build A: ䷝ ䷍ ䷁ ䷱ ䷾ ䷿ ䷌ ䷊ ䷀
    Build B: ䷲ ䷏ ䷧ ䷟ ䷶ ䷵ ䷐ ䷮ ䷊
    → 完全不同·肉眼无法看出"只差一个卦"
```

### 三、加盐——防止碰撞

```
hash = SHA256(Build数据 + 固定盐值) → 取前54bit

盐值 = "星河倒悬·卦码v2.0" 的UTF-8字节

为什么加盐：
  - 防止两个不同配置偶然哈希到同一个结果（碰撞概率<1/2^54）
  - 防止玩家逆向工程（没有盐值→无法从卦码反推Build配置）
  - 盐值不公开 → 只有游戏能解码
```

### 四、哈希→卦符映射

```
哈希输出的54bit → 切成9段·每段6bit → 每段值范围0-63

映射表（值→卦符·按京房八宫序）：
  0=䷀乾  1=䷫姤  2=䷠遁  3=䷋否  4=䷓观  5=䷖剥  6=䷢晋  7=䷍大有
  8=䷁坤  9=䷗复 10=䷒临 11=䷊泰 12=䷡大壮13=䷪夬14=䷄需 15=䷇比
  ...(共64个·按八宫卦序排列·见§附录)

不需要第9卦做校验——哈希本身就是校验。接收方重新哈希对比即可。
```

### 五、解码流程

```
收到卦码 → 9个卦符 → 每个卦符→查表→6bit → 拼成54bit

但无法从这54bit反推Build配置（哈希是单向的）！

解码方式：游戏本地存了一个"卦码↔Build"的映射表

分享流程：
  1. 玩家A生成卦码 → 游戏本地计算hash → 存储(hash, Build配置)
  2. 玩家A分享卦码（9个卦符）
  3. 玩家B导入卦码 → 游戏把卦码发送到服务器
  4. 服务器查表：这个hash对应的Build配置是什么？
  5. 服务器返回Build配置 → 玩家B一键装载

或者离线方式：
  卦码本身就编码了配置+哈希校验
  接收方只需验证：重新哈希配置 → 对比卦码 → 一致=有效
```

### 六、实际卦码示例

```
焦烈·火系专精（终章）：
  ䷝ ䷍ ䷁ ䷱ ䷾ ䷿ ䷌ ䷊ ䷀

同一Build·改一个下卦（乾→坤）：
  ䷲ ䷏ ䷧ ䷟ ䷶ ䷵ ䷐ ䷮ ䷊
  → 完全不同！这就是改枪码效果

新手刚出星纪宫：
  ䷁ ䷁ ䷁ ䷁ ䷁ ䷁ ䷁ ䷁ ䷊
  → 大量䷁（巧合·哈希恰好产出大量坤卦·不会始终如此）
  → 实际上新手码也会是随机分布·无法从卦符密度判断Build成熟度

同一个玩家·终章·同一个Build重新生成：
  → 始终一样（哈希是确定性的·相同输入=相同输出）
```

### 七、v1.0 vs v2.0

| | v1.0 | v2.0 |
|------|------|------|
| 长度 | 52字符 | 9卦一排 |
| 编码 | 结构化·每段=固定含义 | 哈希·改一处全码皆变 |
| 可逆 | ✅ 可解码回Build | ❌ 单向·需映射表 |
| 视觉 | 密集·像序列号 | 松散·像占卜结果 |
| 用途 | 内部存储·精确传输 | 对外分享·社区传播 |
| 碰撞 | 0（结构化编码） | <1/2^54（约1/18亿亿） |
| 改枪码效果 | ❌ 改一处只变一个卦 | ✅ 改一处全码皆变 |

### 八、建议

**两者共存。** v1.0=内部传输协议（精确·可解码）。v2.0=对外分享格式（好看·有改枪码效果）。导入时自动识别格式。

```
游戏内生成卦码 → 同时产出v1.0（存本地）+ v2.0（展示+复制）
社区分享 → 玩家复制v2.0的9个卦符
导入 → 游戏识别9卦格式 → 服务器查表 → 还原Build
```

# v3.0 混合卦码（2026-08-10 新增）
# ══════════════════════════════════════════════

---

## 设计理念

**每个位置可以是八卦（☰）也可以是六十四卦（䷀）。类型由编码算法自动决定——不是玩家选的。**

```
一排固定9个符号。

八卦   = 1bit(类型=0) + 3bit(三爻) = 4bit
六十四卦 = 1bit(类型=1) + 6bit(六爻) = 7bit

类型选择：该位置需要编码多少bit → 自动选类型。
类型信息本身也在编码数据——不是元数据。
```

---

## 一、编码规则

### 1.1 输入

```
Build有效数据 ≈ 47bit：
  上卦4槽（排列编码）   → 11bit
  下卦4槽（排列编码）   → 11bit
  术数3槽 C(12,3)     → 8bit
  法器2槽 C(4,2)      → 3bit
  卦阵2槽 C(8,2)      → 5bit
  同伴2槽 C(12,2)     → 7bit
  版本号               → 2bit

加盐：SHA256(Build数据 + "星河倒悬·卦码v3.0") → 取前54bit
```

### 1.2 切分规则

```
哈希输出54bit → 按4/7bit自适应切分成9个符号：

算法：
  指针=0，符号数组=[]
  循环直到凑满9个符号：
    当前指针位置读取1bit → 决定类型
    如果是0(八卦)：  取后续3bit → 映射到☰☷☲☵☳☴☶☱作为一个符号
    如果是1(六十四卦)：取后续6bit → 映射到䷀-䷿作为一个符号
    指针推进3/6bit
    剩余bit不够 → 用下一个hash块补充

类型bit本身就是数据的一部分——每个符号携带4或7bit有效信息。
```

### 1.3 9个符号必定装满

```
最短情况（9个全是八卦）：9×4=36bit → 36bit不够54bit → 不可能
最长情况（9个全是六十四卦）：9×7=63bit → 超出54bit → 最后截断
典型情况：~5个八卦+4个六十四卦 → 5×4+4×7=48bit ≈ 实际数据量

算法保证：正好用9个符号装完所有bit·最后一个符号用0填充不足的爻线。
```

### 1.4 验证

```
最后一个符号的最后一爻 = 前面所有bit的奇偶校验位。

验证时：
  重新计算前8个符号输出的奇偶 → 与第9符号末爻对比
  一致 → 有效
  不一致 → "此卦不合"
```

## 二、改枪码效果

```
同一个Build → 哈希确定 → 同一个卦码
改一个下卦 → 哈希完全不同 → 9个符号全部改变
两个几乎一样的Build → 码完全不同

类型分布是数据自然形成的结果：
  Build A的数据哈希 → 类型模式 011010110 → 混合5个八卦4个六十四卦
  Build B的数据哈希 → 类型模式 101100101 → 混合4个八卦5个六十四卦
  → 类型分布不同 + 符号内容不同 = 两条完全不同的码
```

## 三、示例

```
构建A·火系专精（终章全Build）：
  ☰ ䷝ ☷ ䷍ ☲ ䷁ ䷱ ☳ ䷊
   ↑   ↑  ↑   ↑  ↑   ↑   ↑  ↑   ↑
   天  火 地 大有 水  晋  鼎 雷  泰
  → 5个八卦+4个六十四卦 = 5×4+4×7 = 48bit有效

同一Build·只改同伴（锻锋→苏羽）：
  ䷲ ䷏ ☰ ䷧ ䷟ ䷶ ☷ ䷵ ☱
  → 7个六十四卦+2个八卦 = 7×7+2×4 = 57bit·类型分布变了
  → 和上一个码完全不同·无法看出"只改了一个同伴"

同一个Build·同一个人重新生成：
  ☰ ䷝ ☷ ䷍ ☲ ䷁ ䷱ ☳ ䷊
  → 和第一次一模一样（哈希确定·相同输入=相同输出）
```

## 四、类型分布不透露Build信息

```
八卦和六十四卦的排列——是哈希输出的结果·不直接对应Build内容。

不是：
  "前面全是六十四卦=终章·前面全是八卦=新手" ← 错误认知

而是：
  类型分布 = hash(Build)的前9bit决定 = 看上去随机
  完全相同的Build可能出5八卦4六十四卦·也可能出3八卦6六十四卦
  → 取决于哈希结果·不由进度决定
  → 避免通过类型分布反推Build信息
```

## 六、建议

**v2.0和v3.0本质相同——都是哈希编码+改枪码效果。v3.0多了一个维度：符号类型自适应。** v2.0是简化实现·v3.0是最终形态。两者可以共存：导入时自动识别。

  混合了八卦和六十四卦 → v3.0格式
  有☷分隔符·长度>20字符 → v1.0格式
